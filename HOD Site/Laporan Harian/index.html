Cara Setup Pantas (Google Sheets + GitHub):

Code.gs (Google Apps Script) 
/**
 * SISTEM LAPORAN HARIAN KERJA LAPANGAN & ADUAN TAPAK (JURUKUR TANAH)
 * Menggunakan Active Spreadsheet serta membina Tab & Header secara automatik jika tiada.
 */

const SHEET_NAME = 'Laporan_Lapangan';
const HEADERS = [
  'Timestamp',
  'Tarikh',
  'Nama',
  'No File',
  'Tugasan / Output',
  'Progress Hari Ini',
  'Total Dah Siap',
  'Jangka Siap Asal',
  'Jangka Siap Semasa',
  'Status',
  'Waktu Mula',
  'Waktu Tamat',
  'Isu / Halangan',
  'Maklumat Tambahan',
  'Perancangan',
  'Gambar',
  'Teks Asal'
];

/**
 * Fungsi pintar untuk mendapatkan Sheet aktif.
 * Membina Tab dan Header secara automatik jika belum wujud.
 */
function getOrCreateSheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(SHEET_NAME);

  // 1. Bina Tab secara automatik jika tiada
  if (!sheet) {
    sheet = ss.insertSheet(SHEET_NAME);
  }

  // 2. Bina Header secara automatik jika tiada atau baris pertama kosong
  const lastRow = sheet.getLastRow();
  let needHeader = false;

  if (lastRow === 0) {
    needHeader = true;
  } else {
    const firstRowValues = sheet.getRange(1, 1, 1, HEADERS.length).getValues()[0];
    const isFirstRowEmpty = firstRowValues.every(cell => String(cell).trim() === '');
    if (isFirstRowEmpty) {
      needHeader = true;
    }
  }

  if (needHeader) {
    const headerRange = sheet.getRange(1, 1, 1, HEADERS.length);
    headerRange.setValues([HEADERS]);
    headerRange.setFontWeight('bold');
    headerRange.setBackground('#0f172a');
    headerRange.setFontColor('#38bdf8');
    sheet.setFrozenRows(1);
  }

  return sheet;
}

/**
 * Menerima penghantaran data laporan baharu daripada Web App (Method: POST)
 */
function doPost(e) {
  const lock = LockService.getScriptLock();
  lock.tryLock(10000);

  try {
    const sheet = getOrCreateSheet();
    let payload = {};

    if (e && e.postData && e.postData.contents) {
      payload = JSON.parse(e.postData.contents);
    } else if (e && e.parameter) {
      payload = e.parameter;
    }

    const rowData = [
      new Date(),
      payload.tarikh || '-',
      payload.nama || '-',
      payload.noFile || '-',
      payload.tugasan || '-',
      payload.progressHariIni || '-',
      payload.totalSiap || '0%',
      payload.jangkaAsal || '-',
      payload.jangkaSemasa || '-',
      payload.status || 'DALAM PROSES',
      payload.waktuMula || '-',
      payload.waktuTamat || '-',
      payload.isu || '-',
      payload.maklumat || '-',
      payload.perancangan || '-',
      payload.image || '',
      payload.rawText || ''
    ];

    sheet.appendRow(rowData);

    return ContentService
      .createTextOutput(JSON.stringify({
        status: 'success',
        message: 'Data berjaya disimpan ke dalam Google Sheet (' + SHEET_NAME + ')'
      }))
      .setMimeType(ContentService.MimeType.JSON);

  } catch (error) {
    return ContentService
      .createTextOutput(JSON.stringify({
        status: 'error',
        message: error.toString()
      }))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

/**
 * Mengambil semula semua rekod daripada Active Sheet untuk dipaparkan di Dashboard (Method: GET)
 */
function doGet(e) {
  try {
    const sheet = getOrCreateSheet();
    const rows = sheet.getDataRange().getDisplayValues();

    if (rows.length <= 1) {
      return ContentService
        .createTextOutput(JSON.stringify({ status: 'success', data: [] }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    const data = rows.slice(1).reverse().map(r => ({
      timestamp: r[0] || '',
      tarikh: r[1] || '-',
      nama: r[2] || '-',
      noFile: r[3] || '-',
      tugasan: r[4] || '-',
      progressHariIni: r[5] || '-',
      totalSiap: r[6] || '0%',
      jangkaAsal: r[7] || '-',
      jangkaSemasa: r[8] || '-',
      status: r[9] || 'DALAM PROSES',
      waktuMula: r[10] || '-',
      waktuTamat: r[11] || '-',
      isu: r[12] || '-',
      maklumat: r[13] || '-',
      perancangan: r[14] || '-',
      image: r[15] || '',
      rawText: r[16] || ''
    }));

    return ContentService
      .createTextOutput(JSON.stringify({ status: 'success', data: data }))
      .setMimeType(ContentService.MimeType.JSON);

  } catch (error) {
    return ContentService
      .createTextOutput(JSON.stringify({ status: 'error', message: error.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
