Cara Setup Pantas (Google Sheets + GitHub):
  1. Buka Google Sheets anda → klik menu Extensions > Apps Script.
  2. Padam semua kod asal dalam Code.gs dan tampal kod penuh di bawah.
  3. Klik Deploy > New deployment → pilih jenis Web app.
  4. Set Execute as: Me dan Who has access: Anyone → klik Deploy dan salin Web App URL.
  5. Buka fail index.html dan gantikan nilai const MACRO_URL = "..."; sebelum push ke GitHub.
  6. Kali pertama deploy v2.8, Google akan minta kebenaran Google Drive. Klik Allow (untuk simpan gambar dalam folder Gambar_Laporan_Lapangan).
  7. Ada gambar lama dalam Sheet? Dalam editor Apps Script, pilih fungsi migrateImagesToDrive → Run sekali. Boleh run semula jika belum habis.
  8. Bila update kod sedia ada: klik Deploy > Manage deployments → ikon pensel → Version: New version → Deploy. URL kekal sama. Kalau skip langkah ni, URL masih jalankan kod lama.
Code.gs (Google Apps Script) 
/**
 * SISTEM LAPORAN HARIAN KERJA LAPANGAN & ADUAN TAPAK (JURUKUR TANAH)
 * Menggunakan Active Spreadsheet serta membina Tab & Header secara automatik jika tiada.
 * v2.7: Fungsi EDIT laporan (ikut ID + semakan versi) dan Log_Edit untuk jejak audit.
 * v2.8: Gambar disimpan sebagai fail dalam Google Drive. Sheet hanya simpan link.
 *       Jalankan migrateImagesToDrive() sekali untuk pindah gambar lama (Base64) ke Drive.
 */

const SHEET_NAME = 'Laporan_Lapangan';
const LOG_SHEET_NAME = 'Log_Edit';

// Folder Drive untuk gambar. Biarkan DRIVE_FOLDER_ID kosong untuk bina folder automatik
// bernama DRIVE_FOLDER_NAME. Jika mahu guna folder sedia ada, tampal ID folder di sini.
const DRIVE_FOLDER_ID = '';
const DRIVE_FOLDER_NAME = 'Gambar_Laporan_Lapangan';

// true = fail dikongsi "Anyone with the link (Viewer)" supaya gambar boleh dipapar di dashboard
const SHARE_IMAGES_PUBLIC = true;

const HEADERS = [
  'Timestamp',
  'Tarikh',
  'Nama',
  'No File',
  'Tugasan / Output',
  'Progress Hari Ini',
  'Total Dah Siap',
  'Jangka Siap Asal',
  'Jangka Siap Semasa',
  'Status',
  'Waktu Mula',
  'Waktu Tamat',
  'Isu / Halangan',
  'Maklumat Tambahan',
  'Perancangan',
  'Gambar',
  'Teks Asal',
  'ID',
  'Versi',
  'Kemaskini Terakhir'
];
const LOG_HEADERS = ['Masa Edit', 'ID', 'Versi Lama', 'Data Lama (JSON)'];

// Medan laporan yang boleh diedit (ikut susunan lajur 2 hingga 15)
const FIELD_KEYS = [
  'tarikh', 'nama', 'noFile', 'tugasan', 'progressHariIni', 'totalSiap',
  'jangkaAsal', 'jangkaSemasa', 'status', 'waktuMula', 'waktuTamat',
  'isu', 'maklumat', 'perancangan'
];
const FIELD_DEFAULTS = { totalSiap: '0%', status: 'DALAM PROSES' };

// Nombor lajur (1-based)
const COL = {
  TIMESTAMP: 1,
  FIRST_FIELD: 2,
  TARIKH: 2,
  NAMA: 3,
  NO_FILE: 4,
  GAMBAR: HEADERS.indexOf('Gambar') + 1,
  TEKS_ASAL: HEADERS.indexOf('Teks Asal') + 1,
  ID: HEADERS.indexOf('ID') + 1,
  VERSI: HEADERS.indexOf('Versi') + 1,
  KEMASKINI: HEADERS.indexOf('Kemaskini Terakhir') + 1
};

/**
 * Paksa nilai disimpan sebagai TEKS.
 * - Elak formula injection (nilai bermula dengan = + - @)
 * - Elak Sheets auto-tukar tarikh (12/11/2026) atau peratus (8%) ikut locale
 */
function asText(value) {
  const s = String(value === undefined || value === null ? '' : value);
  return s === '' ? '' : "'" + s;
}

function jsonOutput(obj) {
  return ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

function styleHeader(range) {
  range.setFontWeight('bold');
  range.setBackground('#0f172a');
  range.setFontColor('#38bdf8');
}

/**
 * Folder utama gambar. ID disimpan dalam Script Properties supaya tidak
 * terbina folder berganda dengan nama sama.
 */
function getImageRootFolder() {
  if (DRIVE_FOLDER_ID) return DriveApp.getFolderById(DRIVE_FOLDER_ID);

  const props = PropertiesService.getScriptProperties();
  const savedId = props.getProperty('IMG_FOLDER_ID');
  if (savedId) {
    try {
      const folder = DriveApp.getFolderById(savedId);
      if (!folder.isTrashed()) return folder;
    } catch (e) {
      // Folder dipadam atau tiada akses, bina semula di bawah
    }
  }

  const folder = DriveApp.createFolder(DRIVE_FOLDER_NAME);
  props.setProperty('IMG_FOLDER_ID', folder.getId());
  return folder;
}

// Subfolder ikut bulan (contoh: 2026-11) supaya Drive tersusun
function getMonthFolder(date) {
  const root = getImageRootFolder();
  const name = Utilities.formatDate(date || new Date(), 'Asia/Kuala_Lumpur', 'yyyy-MM');
  const it = root.getFoldersByName(name);
  return it.hasNext() ? it.next() : root.createFolder(name);
}

function safeFileName(text) {
  return String(text || '')
    .replace(/[^A-Za-z0-9\-_.]+/g, '_')
    .replace(/_+/g, '_')
    .replace(/^_|_$/g, '')
    .slice(0, 60) || 'laporan';
}

function isDataImage(value) {
  return /^data:image\/(png|jpe?g|gif|webp);base64,/i.test(String(value || ''));
}

/**
 * Tukar gambar Base64 kepada fail dalam Drive.
 * Pulangkan { url, fileId } dengan url = link "view" Drive untuk disimpan dalam Sheet.
 */
function saveImageToDrive(dataUrl, info) {
  const match = String(dataUrl).match(/^data:(image\/[a-z+]+);base64,(.+)$/i);
  if (!match) throw new Error('Format gambar tidak sah.');

  const mime = match[1].toLowerCase();
  const ext = mime.split('/')[1].replace('jpeg', 'jpg');
  const bytes = Utilities.base64Decode(match[2]);

  const name = [
    safeFileName(info.tarikh),
    safeFileName(info.noFile),
    safeFileName(info.nama),
    String(info.id || '').slice(0, 8)
  ].filter(Boolean).join('_') + '.' + ext;

  const blob = Utilities.newBlob(bytes, mime, name);
  const file = getMonthFolder(new Date()).createFile(blob);
  file.setDescription('Laporan ' + (info.nama || '-') + ' | ' + (info.noFile || '-') + ' | ID ' + (info.id || '-'));

  if (SHARE_IMAGES_PUBLIC) {
    try {
      file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    } catch (e) {
      // Sesetengah akaun Google Workspace sekat perkongsian luar. Fail kekal private.
      console.warn('Gagal kongsi fail ' + file.getId() + ': ' + e);
    }
  }

  return { fileId: file.getId(), url: 'https://drive.google.com/file/d/' + file.getId() + '/view' };
}

function trashDriveFile(fileId) {
  if (!fileId) return;
  try { DriveApp.getFileById(fileId).setTrashed(true); } catch (e) { /* abaikan */ }
}

function driveIdFromUrl(url) {
  const s = String(url || '');
  const m = s.match(/\/d\/([A-Za-z0-9_-]{20,})/) || s.match(/[?&]id=([A-Za-z0-9_-]{20,})/);
  return m ? m[1] : '';
}

/**
 * Sediakan gambar sebelum ditulis ke Sheet:
 * - Base64 -> simpan ke Drive, guna link Drive
 * - URL https -> simpan URL sahaja
 * - Lain-lain -> kosong
 */
function prepareImage(value, info) {
  const v = String(value || '').trim();
  if (!v) return { cell: '', newFileId: '' };
  if (isDataImage(v)) {
    const saved = saveImageToDrive(v, info);
    return { cell: saved.url, newFileId: saved.fileId };
  }
  if (/^https:\/\//i.test(v)) return { cell: v, newFileId: '' };
  return { cell: '', newFileId: '' };
}

/**
 * Untuk dashboard: link Drive ditukar kepada URL thumbnail yang boleh dipapar dalam <img>.
 */
function imageUrlsForClient(cell) {
  const v = String(cell || '').trim();
  if (!v) return { image: '', imageThumb: '', imageLink: '' };
  const id = driveIdFromUrl(v);
  if (id && /drive\.google\.com/i.test(v)) {
    return {
      image: 'https://drive.google.com/thumbnail?id=' + id + '&sz=w1600',
      imageThumb: 'https://drive.google.com/thumbnail?id=' + id + '&sz=w200',
      imageLink: 'https://drive.google.com/file/d/' + id + '/view'
    };
  }
  return { image: v, imageThumb: v, imageLink: /^https:\/\//i.test(v) ? v : '' };
}

/**
 * Dapatkan Sheet utama. Bina Tab & Header jika tiada.
 * Jika header lama (versi sebelum ini) kurang lajur, lajur baharu ditambah di hujung.
 */
function getOrCreateSheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(SHEET_NAME);

  if (!sheet) {
    sheet = ss.insertSheet(SHEET_NAME);
  }

  const headerRange = sheet.getRange(1, 1, 1, HEADERS.length);
  const current = headerRange.getValues()[0];
  let changed = false;

  for (let i = 0; i < HEADERS.length; i++) {
    if (String(current[i]).trim() === '') {
      current[i] = HEADERS[i];
      changed = true;
    }
  }

  if (changed) {
    headerRange.setValues([current]);
    styleHeader(headerRange);
    sheet.setFrozenRows(1);
  }

  return sheet;
}

function getOrCreateLogSheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(LOG_SHEET_NAME);
  if (!sheet) {
    sheet = ss.insertSheet(LOG_SHEET_NAME);
    const range = sheet.getRange(1, 1, 1, LOG_HEADERS.length);
    range.setValues([LOG_HEADERS]);
    styleHeader(range);
    sheet.setFrozenRows(1);
  }
  return sheet;
}

/**
 * Isi ID & Versi untuk rekod lama yang belum ada (rekod sebelum v2.7).
 */
function ensureIds(sheet) {
  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return;

  const range = sheet.getRange(2, COL.ID, lastRow - 1, 2); // ID + Versi
  const values = range.getValues();
  let changed = false;

  values.forEach(v => {
    if (String(v[0]).trim() === '') { v[0] = Utilities.getUuid(); changed = true; }
    if (!Number(v[1])) { v[1] = 1; changed = true; }
  });

  if (changed) range.setValues(values);
}

function findRowById(sheet, id) {
  const lastRow = sheet.getLastRow();
  if (lastRow < 2 || !id) return -1;
  const ids = sheet.getRange(2, COL.ID, lastRow - 1, 1).getDisplayValues();
  const idx = ids.findIndex(r => r[0] === id);
  return idx === -1 ? -1 : idx + 2;
}

/**
 * Tambah laporan baharu. Jika ID sama sudah wujud (contoh: tekan hantar 2 kali),
 * rekod tidak ditambah semula.
 */
function createReport(sheet, payload, imageCell) {
  const id = payload.id;

  if (findRowById(sheet, id) !== -1) {
    return { status: 'success', message: 'Rekod sudah wujud', id: id, duplicate: true };
  }

  const row = [new Date()];
  FIELD_KEYS.forEach(k => row.push(asText(payload[k] || FIELD_DEFAULTS[k] || '-')));
  row.push(asText(imageCell || ''));
  row.push(asText(payload.rawText || ''));
  row.push(id);
  row.push(1);
  row.push('');

  sheet.appendRow(row);
  return { status: 'success', message: 'Data berjaya disimpan ke dalam ' + SHEET_NAME, id: id };
}

/**
 * Kemaskini laporan sedia ada ikut ID.
 * baseVersi mesti sama dengan Versi semasa dalam Sheet. Jika tidak, bermakna
 * orang lain telah edit dahulu, jadi kemaskini ditolak supaya data mereka tidak ditindih.
 * Gambar lama dalam Drive TIDAK dipadam (kekal sebagai rekod), cuma link dalam Sheet ditukar.
 */
function updateReport(sheet, payload, imageCell) {
  const id = String(payload.id || '').trim();
  if (!id) throw new Error('ID laporan tiada.');

  const rowNum = findRowById(sheet, id);
  if (rowNum === -1) throw new Error('Rekod dengan ID ' + id + ' tidak dijumpai.');

  const range = sheet.getRange(rowNum, 1, 1, HEADERS.length);
  const row = range.getValues()[0];
  const currentVersi = Number(row[COL.VERSI - 1]) || 1;

  if (Number(payload.baseVersi) !== currentVersi) {
    throw new Error('Konflik versi: laporan ini telah dikemaskini oleh orang lain. Sync semula dan cuba lagi.');
  }

  // Jejak audit: simpan salinan data lama (gambar disimpan sebagai link sahaja)
  const oldData = {};
  HEADERS.forEach((h, i) => {
    oldData[h] = (h === 'Gambar' && isDataImage(row[i])) ? '[Base64]' : row[i];
  });
  getOrCreateLogSheet().appendRow([
    new Date(),
    id,
    currentVersi,
    JSON.stringify(oldData).slice(0, 49000)
  ]);

  FIELD_KEYS.forEach((k, i) => {
    row[COL.FIRST_FIELD - 1 + i] = asText(payload[k] || FIELD_DEFAULTS[k] || '-');
  });
  if (payload.imageChanged) {
    row[COL.GAMBAR - 1] = asText(imageCell || '');
  }
  if (payload.rawText !== undefined) {
    row[COL.TEKS_ASAL - 1] = asText(payload.rawText || '');
  }
  row[COL.VERSI - 1] = currentVersi + 1;
  row[COL.KEMASKINI - 1] = new Date();

  range.setValues([row]);
  return { status: 'success', message: 'Laporan dikemaskini', id: id, versi: currentVersi + 1 };
}

/**
 * POST: action = 'create' (default) atau 'update'
 * Gambar disimpan ke Drive DAHULU (di luar lock) supaya pengguna lain tidak tersekat
 * menunggu upload. Jika simpan ke Sheet gagal, fail Drive yang baru dibuang ke Trash.
 */
function doPost(e) {
  let payload = {};
  try {
    if (e && e.postData && e.postData.contents) {
      payload = JSON.parse(e.postData.contents);
    } else if (e && e.parameter) {
      payload = e.parameter;
    }
  } catch (err) {
    return jsonOutput({ status: 'error', message: 'Payload tidak sah: ' + err });
  }

  const isUpdate = payload.action === 'update';
  if (!isUpdate) payload.id = String(payload.id || '').trim() || Utilities.getUuid();

  let image = { cell: '', newFileId: '' };
  const needImage = isUpdate ? !!payload.imageChanged : true;
  if (needImage) {
    try {
      image = prepareImage(payload.image, payload);
    } catch (err) {
      return jsonOutput({ status: 'error', message: 'Gagal simpan gambar ke Drive: ' + err });
    }
  }

  const lock = LockService.getScriptLock();
  if (!lock.tryLock(15000)) {
    trashDriveFile(image.newFileId);
    return jsonOutput({ status: 'error', message: 'Server sibuk, cuba lagi sebentar.' });
  }

  try {
    const sheet = getOrCreateSheet();
    const result = isUpdate
      ? updateReport(sheet, payload, image.cell)
      : createReport(sheet, payload, image.cell);

    if (result.duplicate) trashDriveFile(image.newFileId);
    return jsonOutput(result);

  } catch (error) {
    trashDriveFile(image.newFileId);
    return jsonOutput({ status: 'error', message: error.toString() });
  } finally {
    lock.releaseLock();
  }
}

/**
 * GET: pulangkan semua rekod (terbaru dahulu) untuk Dashboard
 */
function doGet(e) {
  const lock = LockService.getScriptLock();
  if (!lock.tryLock(15000)) {
    return jsonOutput({ status: 'error', message: 'Server sibuk, cuba lagi sebentar.' });
  }

  try {
    const sheet = getOrCreateSheet();
    ensureIds(sheet);
    const rows = sheet.getDataRange().getDisplayValues();

    if (rows.length <= 1) {
      return jsonOutput({ status: 'success', data: [] });
    }

    const data = rows.slice(1).reverse().map(r => {
      const img = imageUrlsForClient(r[15]);
      return {
        timestamp: r[0] || '',
        tarikh: r[1] || '-',
        nama: r[2] || '-',
        noFile: r[3] || '-',
        tugasan: r[4] || '-',
        progressHariIni: r[5] || '-',
        totalSiap: r[6] || '0%',
        jangkaAsal: r[7] || '-',
        jangkaSemasa: r[8] || '-',
        status: r[9] || 'DALAM PROSES',
        waktuMula: r[10] || '-',
        waktuTamat: r[11] || '-',
        isu: r[12] || '-',
        maklumat: r[13] || '-',
        perancangan: r[14] || '-',
        image: img.image,
        imageThumb: img.imageThumb,
        imageLink: img.imageLink,
        rawText: r[16] || '',
        id: r[17] || '',
        versi: Number(r[18]) || 1,
        updatedAt: r[19] || ''
      };
    });

    return jsonOutput({ status: 'success', data: data });

  } catch (error) {
    return jsonOutput({ status: 'error', message: error.toString() });
  } finally {
    lock.releaseLock();
  }
}

/**
 * Jalankan fungsi ini SEKALI dari editor Apps Script (pilih migrateImagesToDrive > Run).
 * Setiap sel Gambar yang masih Base64 akan disimpan ke Drive dan digantikan dengan link.
 * Selamat dijalankan berulang kali: sel yang sudah jadi link akan dilangkau.
 * Jika rekod terlalu banyak dan masa tamat (had 6 minit), jalankan semula untuk sambung.
 */
function migrateImagesToDrive() {
  const sheet = getOrCreateSheet();
  const lastRow = sheet.getLastRow();
  if (lastRow < 2) { console.log('Tiada rekod.'); return; }

  ensureIds(sheet);
  const startTime = Date.now();
  const values = sheet.getRange(2, 1, lastRow - 1, HEADERS.length).getDisplayValues();
  let moved = 0, failed = 0, pending = 0;

  for (let i = 0; i < values.length; i++) {
    const r = values[i];
    const cell = r[COL.GAMBAR - 1];
    if (!isDataImage(cell)) continue;

    if (Date.now() - startTime > 5 * 60 * 1000) { pending++; continue; }

    try {
      const saved = saveImageToDrive(cell, {
        tarikh: r[COL.TARIKH - 1],
        nama: r[COL.NAMA - 1],
        noFile: r[COL.NO_FILE - 1],
        id: r[COL.ID - 1]
      });
      sheet.getRange(i + 2, COL.GAMBAR).setValue(asText(saved.url));
      moved++;
    } catch (err) {
      failed++;
      console.warn('Baris ' + (i + 2) + ' gagal: ' + err);
    }
  }

  const msg = 'Migrasi selesai: ' + moved + ' dipindah, ' + failed + ' gagal' +
    (pending ? ', ' + pending + ' belum diproses (jalankan semula fungsi ini)' : '') + '.';
  console.log(msg);
  return msg;
}
